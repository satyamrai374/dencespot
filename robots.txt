# NOTE: includes/config.php has SITE_STAGING = false, so pages emit
# <meta name="robots" content="index, follow">. Setting that constant back to
# true re-applies a site-wide noindex and nothing here will be indexed.
#
# ⚠ A crawler obeys exactly ONE group — the most specific one matching its
# token — and does NOT inherit anything from "User-agent: *". The five AI
# groups below previously carried only "Allow: /", which meant GPTBot,
# PerplexityBot and the rest were permitted to crawl /storage/ (captured leads:
# real patient names, phone numbers and IPs), /includes/ and the enquiry
# handler. Every group therefore repeats the full restriction list. If you add
# a Disallow anywhere, add it to every group.

User-agent: *
Allow: /

# Conversion and handler endpoints — no search value, and indexing the
# thank-you page corrupts conversion measurement.
Disallow: /enquire.php
Disallow: /storage/
Disallow: /includes/

# Query-string variants of real pages (tracking, form errors).
#
# NOT disallowed: ?type= on the results page and ?topic= on the blog. Both
# self-canonicalise to their clean URL, and a URL Google cannot crawl is a URL
# whose canonical Google cannot read — blocking them would strand the signal
# that consolidates them.
Disallow: /*?error=
Disallow: /*?utm_
